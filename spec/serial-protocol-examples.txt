# Starting from the demo data (firmware/station/demo_data.hpp):
#   students S001, S002, S003; items Multimeter (2 units), Oscilloskop (1), Loddestasjon (3)
# the inputs below, in this order, must produce exactly these lines.
# The last two fields are the named item's available and total count.

# B: borrow -> 1 of 2 Multimeter left, still LEDIG
1,LAAN_UT,S001,Multimeter,LEDIG,OK,1,2

2,LAAN_UT,S001,Multimeter,UTLÅNT,OK,0,2

3,LAAN_UT,S001,Multimeter,UTLÅNT,FEIL,0,2
# tilt left: next student
4,VELG,S002,Multimeter,UTLÅNT,OK,0,2
# A: return -> S002 has no loan of Multimeter: refused
5,RETUR,S002,Multimeter,UTLÅNT,FEIL,0,2

6,VELG,S003,Multimeter,UTLÅNT,OK,0,2
7,VELG,S001,Multimeter,UTLÅNT,OK,0,2
